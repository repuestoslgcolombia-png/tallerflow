     7→export function useCustomers(search = '') {
     8→  return useQuery({
     9→    queryKey: ['customers', search],
    10→    queryFn: async () => {
    11→      const res = await fetch(`/api/customers?search=${encodeURIComponent(search)}`)
    12→      if (!res.ok) throw new Error('Error al cargar clientes')
    13→      return res.json()
    14→    },
    15→  })
    16→}

    30→export function useCustomerMutations() {
    31→  const qc = useQueryClient()
    32→  const create = useMutation({
    33→    mutationFn: async (data: any) => {
    34→      const res = await fetch('/api/customers', {
    35→        method: 'POST',
    36→        headers: { 'Content-Type': 'application/json' },
    37→        body: JSON.stringify(data),
    38→      })
    39→      if (!res.ok) {
    40→        const err = await res.json()
    41→        throw new Error(err.error || 'Error al crear cliente')
    42→      }
    43→      return res.json()
    44→    },
    45→    onSuccess: () => {
    46→      qc.invalidateQueries({ queryKey: ['customers'] })
    47→      toast.success('Cliente creado exitosamente')
    48→    },
    49→    onError: (e: Error) => toast.error(e.message),
    50→  })
    51→  const update = useMutation({
    52→    mutationFn: async ({ id, data }: { id: string; data: any }) => {
    53→      const res = await fetch(`/api/customers/${id}`, {
    54→        method: 'PUT',
    55→        headers: { 'Content-Type': 'application/json' },
    56→        body: JSON.stringify(data),
    57→      })
    58→      if (!res.ok) {
    59→        const err = await res.json()
    60→        throw new Error(err.error || 'Error al actualizar')
    61→      }
    62→      return res.json()
    63→    },
    64→    onSuccess: () => {
    65→      qc.invalidateQueries({ queryKey: ['customers'] })
    66→      qc.invalidateQueries({ queryKey: ['customer'] })
    67→      toast.success('Cliente actualizado')
    68→    },
    69→    onError: (e: Error) => toast.error(e.message),
    70→  })
    71→  const remove = useMutation({
    72→    mutationFn: async (id: string) => {
    73→      const res = await fetch(`/api/customers/${id}`, { method: 'DELETE' })
    74→      if (!res.ok) {
    75→        const err = await res.json()
    76→        throw new Error(err.error || 'Error al eliminar')
    77→      }
    78→      return res.json()
    79→    },
    80→    onSuccess: () => {
    81→      qc.invalidateQueries({ queryKey: ['customers'] })
    82→      toast.success('Cliente eliminado')
    83→    },
    84→    onError: (e: Error) => toast.error(e.message),
    85→  })
    86→  return { create, update, remove }
    87→}

   616→export function useInvoiceMutations() {
   617→  const qc = useQueryClient()
   618→  const create = useMutation({
   619→    mutationFn: async (data: any) => {
   620→      const res = await fetch('/api/invoices', {
   621→        method: 'POST',
   622→        headers: { 'Content-Type': 'application/json' },
   623→        body: JSON.stringify(data),
   624→      })
   625→      if (!res.ok) {
   626→        const err = await res.json()
   627→        throw new Error(err.error || 'Error al crear factura')
   628→      }
   629→      return res.json()
   630→    },
   631→    onSuccess: () => {
   632→      qc.invalidateQueries({ queryKey: ['invoices'] })
   633→      qc.invalidateQueries({ queryKey: ['work-orders'] })
   634→      qc.invalidateQueries({ queryKey: ['work-order'] })
   635→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   636→      toast.success('Factura creada exitosamente')
   637→    },
   638→    onError: (e: Error) => toast.error(e.message),
   639→  })
   640→  const update = useMutation({
   641→    mutationFn: async ({ id, data }: { id: string; data: any }) => {
   642→      const res = await fetch(`/api/invoices/${id}`, {
   643→        method: 'PUT',
   644→        headers: { 'Content-Type': 'application/json' },
   645→        body: JSON.stringify(data),
   646→      })
   647→      if (!res.ok) {
   648→        const err = await res.json()
   649→        throw new Error(err.error || 'Error al actualizar factura')
   650→      }
   651→      return res.json()
   652→    },
   653→    onSuccess: () => {
   654→      qc.invalidateQueries({ queryKey: ['invoices'] })
   655→      qc.invalidateQueries({ queryKey: ['invoice'] })
   656→      qc.invalidateQueries({ queryKey: ['work-orders'] })
   657→      qc.invalidateQueries({ queryKey: ['work-order'] })
   658→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   659→      toast.success('Factura actualizada')
   660→    },
   661→    onError: (e: Error) => toast.error(e.message),
   662→  })
   663→  const remove = useMutation({
   664→    mutationFn: async (id: string) => {
   665→      const res = await fetch(`/api/invoices/${id}`, { method: 'DELETE' })
   666→      if (!res.ok) {
   667→        const err = await res.json()
   668→        throw new Error(err.error || 'Error al eliminar factura')
   669→      }
   670→      return res.json()
   671→    },
   672→    onSuccess: () => {
   673→      qc.invalidateQueries({ queryKey: ['invoices'] })
   674→      toast.success('Factura eliminada')
   675→    },
   676→    onError: (e: Error) => toast.error(e.message),
   677→  })
   678→  return { create, update, remove }
   679→}

   810→// Helper para generar URL de WhatsApp (wa.me)
   811→export function buildWhatsAppUrl(phone: string, message: string): string {
   812→  const cleanPhone = phone.replace(/[^0-9]/g, '')
   813→  const encodedMessage = encodeURIComponent(message)
   814→  return `https://wa.me/${cleanPhone}?text=${encodedMessage}`
   815→}

   817→// Helper para enviar un mensaje por WhatsApp (abre wa.me) y registrar el envío
   818→export async function sendWhatsAppMessage(params: {
   819→  phone: string
   820→  message: string
   821→  customerId: string
   822→  workOrderId?: string
   823→  reminderId?: string
   824→  templateId?: string
   825→  customerName?: string
   826→}) {
   827→  const url = buildWhatsAppUrl(params.phone, params.message)
   828→  if (typeof window !== 'undefined') {
   829→    window.open(url, '_blank', 'noopener,noreferrer')
   830→  }
   831→  try {
   832→    await fetch('/api/whatsapp/messages', {
   833→      method: 'POST',
   834→      headers: { 'Content-Type': 'application/json' },
   835→      body: JSON.stringify({
   836→        customerId: params.customerId,
   837→        workOrderId: params.workOrderId || null,
   838→        reminderId: params.reminderId || null,
   839→        templateId: params.templateId || null,
   840→        toPhone: params.phone,
   841→        toName: params.customerName,
   842→        message: params.message,
   843→        status: 'sent',
   844→        channel: 'whatsapp',
   845→        sentBy: 'Usuario',
   846→      }),
   847→    })
   848→  } catch {
   849→    // best-effort
   850→  }
   851→  return url
   852→}
