   452→   452→function PartFormDialog({
   453→   453→  open,
   454→   454→  onOpenChange,
   455→   455→  mode,
   456→   456→  part,
   457→   457→}: {
   458→   458→  open: boolean
   459→   459→  onOpenChange: (v: boolean) => void
   460→   460→  mode: 'create' | 'edit'
   461→   461→  part?: any | null
   462→   462→}) {
   463→   463→  const { create, update } = usePartMutations()
   464→   464→
   465→   465→  const [sku, setSku] = React.useState('')
   466→   466→  const [name, setName] = React.useState('')
   467→   467→  const [description, setDescription] = React.useState('')
   468→   468→  const [category, setCategory] = React.useState('')
   469→   469→  const [unit, setUnit] = React.useState('unidad')
   470→   470→  const [stock, setStock] = React.useState('0')
   471→   471→  const [minStock, setMinStock] = React.useState('0')
   472→   472→  const [unitCost, setUnitCost] = React.useState('0')
   473→   473→  const [unitPrice, setUnitPrice] = React.useState('0')
   474→   474→  const [location, setLocation] = React.useState('')
   475→   475→
   476→   476→  React.useEffect(() => {
   477→   477→    if (open) {
   478→   478→      if (mode === 'edit' && part) {
   479→   479→        setSku(part.sku || '')
   480→   480→        setName(part.name || '')
   481→   481→        setDescription(part.description || '')
   482→   482→        setCategory(part.category || '')
   483→   483→        setUnit(part.unit || 'unidad')
   484→   484→        setStock(String(part.stock ?? 0))
   485→   485→        setMinStock(String(part.minStock ?? 0))
   486→   486→        setUnitCost(String(part.unitCost ?? 0))
   487→   487→        setUnitPrice(String(part.unitPrice ?? 0))
   488→   488→        setLocation(part.location || '')
   489→   489→      } else {
   490→   490→        setSku('')
   491→   491→        setName('')
   492→   492→        setDescription('')
   493→   493→        setCategory('')
   494→   494→        setUnit('unidad')
   495→   495→        setStock('0')
   496→   496→        setMinStock('0')
   497→   497→        setUnitCost('0')
   498→   498→        setUnitPrice('0')
   499→   499→        setLocation('')
   500→   500→      }
   501→   501→    }
   502→   502→  }, [open, mode, part])
   503→   503→

   703→   703→function AdjustStockDialog({
   704→   704→  part,
   705→   705→  onOpenChange,
   706→   706→}: {
   707→   707→  part: any | null
   708→   708→  onOpenChange: (v: boolean) => void
   709→   709→}) {
   710→   710→  const { update } = usePartMutations()
   711→   711→  const [quantity, setQuantity] = React.useState('')
   712→   712→  const [movementType, setMovementType] = React.useState<string>('in')
   713→   713→  const [reason, setReason] = React.useState('')
   714→   714→
   715→   715→  React.useEffect(() => {
   716→   716→    if (part) {
   717→   717→      setQuantity('')
   718→   718→      setReason('')
   719→   719→      // default movement type based on nothing yet — let user enter qty first
   720→   720→      setMovementType('in')
   721→   721→    }
   722→   722→  }, [part])
   723→   723→
   724→   724→  // Auto-suggest movement type from sign of quantity
   725→   725→  React.useEffect(() => {
   726→   726→    const n = parseInt(quantity)
   727→   727→    if (!isNaN(n)) {
   728→   728→      if (n > 0) setMovementType('in')
   729→   729→      else if (n < 0) setMovementType('out')
   730→   730→      else setMovementType('adjustment')
   731→   731→    }
   732→   732→  }, [quantity])
   733→   733→
   734→   734→  if (!part) return null
   735→   735→
   736→   736→  const currentStock = part.stock || 0
   737→   737→  const newStock = currentStock + (parseInt(quantity) || 0)
   738→   738→  const canSubmit = quantity !== '' && parseInt(quantity) !== 0
   739→   739→
   740→   740→  const handleSubmit = () => {
   741→   741→    update.mutate(
   742→   742→      {
   743→   743→        id: part.id,
   744→   744→        data: {
   745→   745→          action: 'adjust_stock',
   746→   746→          quantity: parseInt(quantity),
   747→   747→          movementType,
   748→   748→          reason: reason.trim() || undefined,
   749→   749→        },
   750→   750→      },
   751→   751→      { onSuccess: () => onOpenChange(false) }
   752→   752→    )
   753→   753→  }
   754→   754→
   755→   755→  return (
   756→   756→    <Dialog open={!!part} onOpenChange={onOpenChange}>
   757→   757→      <DialogContent>
   758→   758→        <DialogHeader>
   759→   759→          <DialogTitle>Ajustar Stock · {part.sku}</DialogTitle>
   760→   760→          <DialogDescription>
   761→   761→            {part.name}
   762→   762→          </DialogDescription>
   763→   763→        </DialogHeader>

   765→   765→        <div className="space-y-4">
   766→   766→          {/* Current stock indicator */}

   768→   768→            <div>
   769→   769→              <p className="text-xs text-muted-foreground">Stock actual</p>
   770→   770→              <p className="text-2xl font-semibold tabular-nums">{currentStock}</p>
   771→   771→            </div>
   772→   772→            <div className="text-right">
   773→   773→              <p className="text-xs text-muted-foreground">Nuevo stock</p>
   774→   774→              <p
   775→   775→                className={cn(
   776→   776→                  'text-2xl font-semibold tabular-nums',
   777→   777→                  newStock < 0
   778→   778→                    ? 'text-rose-600'
   779→   779→                    : newStock < (part.minStock || 0)
   780→   780→                    ? 'text-amber-600'
   781→   781→                    : 'text-emerald-600'
   782→   782→                )}
   783→   783→              >
   784→   784→                {newStock}
   785→   785→              </p>
   786→   786→            </div>
   787→   787→          </div>
   788→   788→
   789→   789→          <div className="grid grid-cols-2 gap-3">
   790→   790→            <div className="space-y-2">
   791→   791→              <Label>Cantidad (±)</Label>
   792→   792→              <div className="flex gap-2">
   793→   793→                <Button
   794→   794→                  type="button"
   795→   795→                  variant="outline"
   796→   796→                  size="icon"
   797→   797→                  onClick={() => {
   798→   798→                    const n = parseInt(quantity || '0') - 1
   799→   799→                    setQuantity(String(n))
   800→   800→                  }}
   801→   801→                >
   802→   802→                  <ArrowDown className="size-4 text-rose-500" />
   803→   803→                </Button>
