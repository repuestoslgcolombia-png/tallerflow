   238→              <TableHeader>
   239→                <TableRow>
   240→                  <TableHead className="pl-4">SKU</TableHead>
   241→                  <TableHead>Nombre</TableHead>

   296→                            className={cn(
   297→                              'font-medium tabular-nums',
   298→                              stockTone === 'rose' && 'text-rose-600',
   299→                              stockTone === 'amber' && 'text-amber-600',
   300→                              stockTone === 'emerald' && 'text-foreground'
   301→                            )}
   302→                          >
   303→                            {stock}
   304→                          </span>
   305→                          <span className="text-xs text-muted-foreground">{p.unit}</span>
   306→                        </div>
   307→                      </TableCell>
   308→                      <TableCell className="text-center tabular-nums text-muted-foreground">
   309→                        {min}
   310→                      </TableCell>
   311→                      <TableCell className="text-right tabular-nums text-muted-foreground">
   312→                        {formatCurrency(p.unitCost || 0)}
   313→                      </TableCell>
   314→                      <TableCell className="text-right tabular-nums font-medium">
   315→                        {formatCurrency(p.unitPrice || 0)}

   324→                        )}
   325→                      </TableCell>
   326→                      <TableCell className="pr-4 text-right">
   327→                        <PartActionsMenu
   328→                          part={p}
   329→                          onEdit={() => setEditPart(p)}
   330→                          onAdjust={() => setAdjustPart(p)}
   331→                        />
   332→                      </TableCell>
   333→                    </TableRow>
   334→                  )
   335→                })}
   336→              </TableBody>
   337→            </Table>
   338→            </div>
   339→          )}
   340→        </CardContent>
   341→      </Card>
   342→
   343→      {/* Dialogs */}
   344→      <PartFormDialog
   345→        open={createOpen}
   346→        onOpenChange={setCreateOpen}
   347→        mode="create"
   348→      />
   349→      <PartFormDialog
   350→        open={!!editPart}
   351→        onOpenChange={(v) => !v && setEditPart(null)}
   352→        mode="edit"
   353→        part={editPart}
   354→      />
   355→      <AdjustStockDialog
   356→        part={adjustPart}
   357→        onOpenChange={(v) => !v && setAdjustPart(null)}
   358→      />
   359→    </div>
   360→  )
   361→}

   365→function StatCard({
   366→  label,
   367→  value,
   368→  icon,
   369→  tone,
   370→}: {
   371→  label: string
   372→  value: string
   373→  icon: React.ReactNode
   374→  tone: 'slate' | 'rose' | 'emerald' | 'amber' | 'teal'
   375→}) {
   376→  const toneClasses = {
   377→    slate: 'bg-slate-100 text-slate-600',
   378→    rose: 'bg-rose-100 text-rose-700',
   379→    emerald: 'bg-emerald-100 text-emerald-700',
   380→    amber: 'bg-amber-100 text-amber-700',
   381→    teal: 'bg-teal-100 text-teal-700',
   382→  }
   383→  return (
   384→    <Card>
   385→      <CardContent className="p-4">
   386→        <div className="flex items-center justify-between">
   387→          <div className="min-w-0">
   388→            <p className="text-xs text-muted-foreground">{label}</p>
   389→            <p className="truncate text-xl font-semibold tabular-nums">{value}</p>
   390→          </div>
   391→          <div className={cn('flex size-8 items-center justify-center rounded-md', toneClasses[tone])}>
   392→            {icon}
   393→          </div>
   394→        </div>
   395→      </CardContent>
   396→    </Card>
   397→  )
   398→}
   399→
   400→function PartActionsMenu({
   401→  part,
   402→  onEdit,
   403→  onAdjust,
   404→}: {
   405→  part: any
   406→  onEdit: () => void
   407→  onAdjust: () => void
   408→}) {
   409→  const { update } = usePartMutations()
   410→
   411→  const toggleActive = () => {
   412→    update.mutate({
   413→      id: part.id,
   414→      data: { active: !part.active },
   415→    })
   416→  }
   417→
   418→  return (
   419→    <div className="flex justify-end">
   420→      <DropdownMenu>
   421→        <DropdownMenuTrigger asChild>
   422→          <Button variant="ghost" size="icon" className="size-8">
   423→            <MoreHorizontal className="size-4" />
   424→          </Button>
   425→        </DropdownMenuTrigger>
   426→        <DropdownMenuContent align="end" className="w-44">
   427→          <DropdownMenuItem onClick={onAdjust}>
   428→            <Settings className="size-4" /> Ajustar stock
   429→          </DropdownMenuItem>
   430→          <DropdownMenuItem onClick={onEdit}>
   431→            <Pencil className="size-4" /> Editar
   432→          </DropdownMenuItem>
   433→          <DropdownMenuSeparator />
   434→          <DropdownMenuItem onClick={toggleActive} disabled={update.isPending}>
   435→            {part.active ? (
   436→              <>
   437→                <X className="size-4" /> Desactivar
   438→              </>
   439→            ) : (
   440→              <>
   441→                <Package className="size-4" /> Activar
   442→              </>
   443→            )}
   444→          </DropdownMenuItem>
   445→        </DropdownMenuContent>
   446→      </DropdownMenu>
   447→    </div>
   448→  )
   449→}

   451→// ============== Part Form Dialog (create/edit) ==============
   452→function PartFormDialog({
   453→  open,
   454→  onOpenChange,
   455→  mode,
   456→  part,
   457→}: {
   458→  open: boolean
   459→  onOpenChange: (v: boolean) => void
   460→  mode: 'create' | 'edit'
   461→  part?: any | null
   462→}) {
   463→  const { create, update } = usePartMutations()
   464→
   465→  const [sku, setSku] = React.useState('')
   466→  const [name, setName] = React.useState('')
   467→  const [description, setDescription] = React.useState('')
   468→  const [category, setCategory] = React.useState('')
   469→  const [unit, setUnit] = React.useState('unidad')
   470→  const [stock, setStock] = React.useState('0')
   471→  const [minStock, setMinStock] = React.useState('0')
   472→  const [unitCost, setUnitCost] = React.useState('0')
   473→  const [unitPrice, setUnitPrice] = React.useState('0')
   474→  const [location, setLocation] = React.useState('')
   475→
   476→  React.useEffect(() => {
   477→    if (open) {
   478→      if (mode === 'edit' && part) {
   479→        setSku(part.sku || '')
   480→        setName(part.name || '')
   481→        setDescription(part.description || '')
   482→        setCategory(part.category || '')
   483→        setUnit(part.unit || 'unidad')
   484→        setStock(String(part.stock ?? 0))
   485→        setMinStock(String(part.minStock ?? 0))
   486→        setUnitCost(String(part.unitCost ?? 0))
   487→        setUnitPrice(String(part.unitPrice ?? 0))
   488→        setLocation(part.location || '')
   489→      } else {
   490→        setSku('')
   491→        setName('')
   492→        setDescription('')
   493→        setCategory('')
   494→        setUnit('unidad')
   495→        setStock('0')
   496→        setMinStock('0')
   497→        setUnitCost('0')
   498→        setUnitPrice('0')
   499→        setLocation('')
   500→      }
   501→    }
   502→  }, [open, mode, part])
   503→
   504→  const canSubmit = sku.trim() && name.trim()
   505→
   506→  const handleSubmit = () => {
   507→    const data: any = {
   508→      sku: sku.trim(),
   509→      name: name.trim(),
   510→      description: description.trim() || undefined,
   511→      category: category.trim() || undefined,
   512→      unit: unit || 'unidad',
   513→      minStock: parseInt(minStock) || 0,
   514→      unitCost: parseFloat(unitCost) || 0,
   515→      unitPrice: parseFloat(unitPrice) || 0,
   516→      location: location.trim() || undefined,
   517→    }
   518→    if (mode === 'create') {
   519→      data.stock = parseInt(stock) || 0
   520→      create.mutate(data, { onSuccess: () => onOpenChange(false) })
   521→    } else {
   522→      update.mutate(
   523→        { id: part.id, data },
   524→        { onSuccess: () => onOpenChange(false) }
   525→      )
   526→    }
   527→  }
   528→
   529→  const isPending = mode === 'create' ? create.isPending : update.isPending
   530→
   531→  return (
   532→    <Dialog open={open} onOpenChange={onOpenChange}>
   533→      <DialogContent className="sm:max-w-2xl">
   534→        <DialogHeader>
   535→          <DialogTitle>
   536→            {mode === 'create' ? 'Nuevo Repuesto' : `Editar ${part?.sku}`}
   537→          </DialogTitle>
   538→          <DialogDescription>
   539→            {mode === 'create'
   540→              ? 'Registra un nuevo repuesto o material en el inventario.'
   541→              : 'Modifica la información del repuesto.'}
   542→          </DialogDescription>
   543→        </DialogHeader>
   544→
   545→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   546→          <div className="space-y-2">
   547→            <Label>
   548→              SKU <span className="text-rose-500">*</span>
   549→            </Label>
   550→            <div className="relative">
   551→              <Hash className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   552→              <Input
   553→                className="pl-9 font-mono"
   554→                value={sku}
   555→                onChange={(e) => setSku(e.target.value)}
   556→                placeholder="Ej: RAM-8GB-DDR4"
   557→              />
   558→            </div>
   559→          </div>
   560→
   561→          <div className="space-y-2">
   562→            <Label>
   563→              Nombre <span className="text-rose-500">*</span>
   564→            </Label>
   565→            <Input
   566→              value={name}
   567→              onChange={(e) => setName(e.target.value)}
   568→              placeholder="Ej: Memoria RAM 8GB DDR4"
   569→            />
   570→          </div>
   571→
   572→          <div className="space-y-2 sm:col-span-2">
   573→            <Label>Descripción</Label>
   574→            <Textarea
   575→              value={description}
   576→              onChange={(e) => setDescription(e.target.value)}
   577→              rows={2}
   578→              placeholder="Detalles técnicos, marca, compatibilidad…"
   579→            />
   580→          </div>
   581→
   582→          <div className="space-y-2">
   583→            <Label>Categoría</Label>
   584→            <Input
   585→              value={category}
   586→              onChange={(e) => setCategory(e.target.value)}
   587→              placeholder="RAM, SSD, Pantalla, Batería…"
   588→              list="part-categories"
   589→            />
   590→            <datalist id="part-categories">
   591→              <option value="RAM" />
   592→              <option value="SSD" />
   593→              <option value="HDD" />
   594→              <option value="Pantalla" />
   595→              <option value="Batería" />
   596→              <option value="Cable" />
   597→              <option value="Fuente" />
   598→              <option value="Otro" />
   599→            </datalist>
   600→          </div>
   601→
   602→          <div className="space-y-2">
   603→            <Label>Unidad</Label>
   604→            <Select value={unit} onValueChange={setUnit}>
   605→              <SelectTrigger className="w-full">
   606→                <SelectValue />
   607→              </SelectTrigger>
   608→              <SelectContent>
   609→                <SelectItem value="unidad">unidad</SelectItem>
   610→                <SelectItem value="metro">metro</SelectItem>

   702→// ============== Adjust Stock Dialog ==============
   703→function AdjustStockDialog({
   704→  part,
   705→  onOpenChange,
   706→}: {
   707→  part: any | null
   708→  onOpenChange: (v: boolean) => void
   709→}) {
   710→  const { update } = usePartMutations()
   711→  const [quantity, setQuantity] = React.useState('')
   712→  const [movementType, setMovementType] = React.useState<string>('in')
   713→  const [reason, setReason] = React.useState('')
   714→
   715→  React.useEffect(() => {
   716→    if (part) {
   717→      setQuantity('')
   718→      setReason('')
   719→      // default movement type based on nothing yet — let user enter qty first
   720→      setMovementType('in')
   721→    }
   722→  }, [part])
   723→
   724→  // Auto-suggest movement type from sign of quantity
   725→  React.useEffect(() => {
   726→    const n = parseInt(quantity)
   727→    if (!isNaN(n)) {
   728→      if (n > 0) setMovementType('in')
   729→      else if (n < 0) setMovementType('out')
   730→      else setMovementType('adjustment')
   731→    }
   732→  }, [quantity])
   733→
   734→  if (!part) return null
   735→
   736→  const currentStock = part.stock || 0
   737→  const newStock = currentStock + (parseInt(quantity) || 0)
   738→  const canSubmit = quantity !== '' && parseInt(quantity) !== 0
   739→
   740→  const handleSubmit = () => {
   741→    update.mutate(
   742→      {
   743→        id: part.id,
   744→        data: {
   745→          action: 'adjust_stock',
   746→          quantity: parseInt(quantity),
   747→          movementType,
   748→          reason: reason.trim() || undefined,
   749→        },
   750→      },
   751→      { onSuccess: () => onOpenChange(false) }
   752→    )
   753→  }
   754→
   755→  return (
   756→    <Dialog open={!!part} onOpenChange={onOpenChange}>
   757→      <DialogContent>
   758→        <DialogHeader>
   759→          <DialogTitle>Ajustar Stock · {part.sku}</DialogTitle>
   760→          <DialogDescription>
   761→            {part.name}
   762→          </DialogDescription>
   763→        </DialogHeader>
   764→
   765→        <div className="space-y-4">
   766→          {/* Current stock indicator */}
   767→          <div className="flex items-center justify-between rounded-md border bg-muted/30 p-3">
   768→            <div>
   769→              <p className="text-xs text-muted-foreground">Stock actual</p>
   770→              <p className="text-2xl font-semibold tabular-nums">{currentStock}</p>
   771→            </div>
   772→            <div className="text-right">
   773→              <p className="text-xs text-muted-foreground">Nuevo stock</p>
   774→              <p
   775→                className={cn(
   776→                  'text-2xl font-semibold tabular-nums',
   777→                  newStock < 0
   778→                    ? 'text-rose-600'
   779→                    : newStock < (part.minStock || 0)
   780→                    ? 'text-amber-600'
   781→                    : 'text-emerald-600'
   782→                )}
   783→              >
   784→                {newStock}
   785→              </p>
   786→            </div>
   787→          </div>
   788→
   789→          <div className="grid grid-cols-2 gap-3">
   790→            <div className="space-y-2">
   791→              <Label>Cantidad (±)</Label>
   792→              <div className="flex gap-2">
   793→                <Button
   794→                  type="button"
   795→                  variant="outline"
   796→                  size="icon"
   797→                  onClick={() => {
   798→                    const n = parseInt(quantity || '0') - 1
   799→                    setQuantity(String(n))
   800→                  }}
   801→                >
   802→                  <ArrowDown className="size-4 text-rose-500" />
   803→                </Button>
   804→                <Input
   805→                  type="number"
   806→                  value={quantity}
   807→                  onChange={(e) => setQuantity(e.target.value)}
   808→                  placeholder="0"
   809→                  className="text-center tabular-nums"
   810→                />
   811→                <Button
   812→                  type="button"
   813→                  variant="outline"
   814→                  size="icon"
   815→                  onClick={() => {
   816→                    const n = parseInt(quantity || '0') + 1
   817→                    setQuantity(String(n))
   818→                  }}
   819→                >
   820→                  <ArrowUp className="size-4 text-emerald-500" />
   821→                </Button>
   822→              </div>
   823→            </div>
   824→
   825→            <div className="space-y-2">
   826→              <Label>Tipo de movimiento</Label>
   827→              <Select value={movementType} onValueChange={setMovementType}>
   828→                <SelectTrigger className="w-full">
   829→                  <SelectValue />
   830→                </SelectTrigger>
   831→                <SelectContent>
   832→                  {(Object.keys(MOVEMENT_TYPES) as (keyof typeof MOVEMENT_TYPES)[]).map((k) => (
   833→                    <SelectItem key={k} value={k}>
   834→                      {MOVEMENT_TYPES[k].label}
   835→                    </SelectItem>
   836→                  ))}
   837→                </SelectContent>
   838→              </Select>
   839→            </div>
   840→          </div>
   841→
   842→          <div className="space-y-2">
   843→            <Label>Motivo (opcional)</Label>
   844→            <Textarea
   845→              value={reason}
   846→              onChange={(e) => setReason(e.target.value)}
   847→              rows={2}
   848→              placeholder="Compra, merma, ajuste de inventario, uso en OT…"
   849→            />
   850→          </div>
   851→        </div>
   852→
   853→        <DialogFooter>
   854→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   855→            Cancelar
   856→          </Button>
   857→          <Button
   858→            onClick={handleSubmit}
   859→            disabled={!canSubmit || update.isPending}
   860→            className="gap-2"
   861→          >
   862→            {update.isPending && <Loader2 className="size-4 animate-spin" />}
   863→            Aplicar Ajuste
   864→          </Button>
   865→        </DialogFooter>
   866→      </DialogContent>
   867→    </Dialog>
   868→  )
   869→}
